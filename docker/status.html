How to check from a terminal

docker ps                                   # is the library container up?
docker/run.sh --list                        # installed titles
curl -s http://127.0.0.1:8123/ >/dev/null && echo running || echo "not running"

How to start / sync / get the files

# start the library (menu + this API) — no key needed once synced
docker/run.sh --library

# pull / update the collection (needs an itch.io API key)
ITCH_API_KEY=... docker/run.sh --library --sync

# copy the game files to your machine
docker run --rm -v arcade-contract-sim-games:/games \
  -v "$PWD/games-export:/out" alpine \
  sh -c 'mkdir -p /out && cp -r /games/web /out/'
